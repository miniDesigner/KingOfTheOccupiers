  _pending = null;

      } catch (e2) {

/**

 * 目标：同一对局内所有随机决策必须来自同一随机源，保证双端按相同命令序列

 *

/**

export function createRng(seed) {
  let a = seed >>> 0;
  return function next() {
    a |= 0;
    a = (a + 0x6D2B79F5) | 0;

    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
